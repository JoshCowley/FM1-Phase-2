# After a note is deleted in NOTES (EDIT), the cursor goes on to the next step; at the pattern's last step it wraps to the first
# (as entering a note does). A lasso's delete goes on to the step after the lasso. Nothing picked: nothing happens, the cursor stays.
wait 500
tap SEQ
wait 300
key w7 down 100
wait 60
key w7 up
key w9 down 100
wait 60
key w9 up
key w11 down 100
wait 60
key w11 up
key w8 down 100
wait 60
key w8 up
wait 300
expect seq_cursor 4
expect seq_cnt0 1
expect seq_cnt3 1
turn ALGO -1
wait 200
expect seq_cursor 0
tap EDIT
wait 300
expect seq_cnt0 0
expect seq_cursor 1
tap EDIT
wait 300
expect seq_cnt1 0
expect seq_cursor 2
tap EDIT
wait 300
expect seq_cursor 3
tap EDIT
wait 300
expect seq_cnt3 0
expect seq_cursor 4
tap EDIT
wait 300
expect seq_cursor 4
